FIRESTORE SECURITY RULES:
=========================
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // User profile and all private subcollections under users/{userId}
    match /users/{userId} {
      // Allow user to read and write their own profile document
      // Also allow authenticated users to read public lawyer profiles
      allow read: if request.auth != null && (request.auth.uid == userId || resource.data.role == 'lawyer');
      allow write: if request.auth != null && request.auth.uid == userId;

      // Subcollections: learningHistory, problemHistory, documents, bookmarks, settings, chatSessions
      match /learningHistory/{historyId} {
        allow read, write: if request.auth != null && request.auth.uid == userId;
      }

      match /problemHistory/{sessionId} {
        allow read, write: if request.auth != null && request.auth.uid == userId;
        
        match /messages/{messageId} {
          allow read, write: if request.auth != null && request.auth.uid == userId;
        }
      }

      match /documents/{documentId} {
        allow read, write: if request.auth != null && request.auth.uid == userId;
      }

      match /bookmarks/{bookmarkId} {
        allow read, write: if request.auth != null && request.auth.uid == userId;
      }

      match /settings/{settingId} {
        allow read, write: if request.auth != null && request.auth.uid == userId;
      }

      match /chatSessions/{sessionId} {
        allow read, write: if request.auth != null && request.auth.uid == userId;
        
        match /messages/{messageId} {
          allow read, write: if request.auth != null && request.auth.uid == userId;
        }
      }

      // Consultations subcollection (accessible by the owner and the counterpart lawyer/user)
      match /consultations/{consultationId} {
        allow read, write: if request.auth != null;
      }
    }
  }
}

STORAGE SECURITY RULES:
=======================
rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    match /profiles/{userId}/{allPaths=**} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }
    match /consultations/{userId}/{allPaths=**} {
      allow read, write: if request.auth != null;
    }
  }
}
